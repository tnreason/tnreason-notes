    \section{Tensor networks for neuro-symbolic AI}

    \begin{frame}{Motivation: Toy accounting rule system}

        \begin{block}{Toy accounting rule system}
            Assume you want to book one of two accounts $\mathrm{A1}$ and $\mathrm{A2}$ depending on a feature $\mathrm{F}$ of an invoice, based on two rules:
            \begin{itemize}
                \item \textcolor{\concolor}{Exactly one account must be booked.}
                \item \textcolor{\probcolor}{If feature $\mathrm{F}$ is present on the invoice, the account $\mathrm{A1}$ is typically booked.}
            \end{itemize}
        \end{block}

        \begin{itemize}
            \item {Accounting rules} are formalised in propositional logics, triggered by receipts properties.
            \item These rules can either be \textcolor{\concolor}{hard restrictions}, or \textcolor{\probcolor}{prone to exceptions}.
        \end{itemize}

        \begin{center}
            \bf We apply tensor networks for neuro-symbolic use cases like this!
        \end{center}

    \end{frame}

    \begin{frame}{Tensors: Natural description of factored representations}

        {\bf Factored representations}
        are collection of variables $\shortcatvariables$, each with a finite set of states $[\catdimof{\catenumerator}]=\{0,\ldots,\catdimof{\catenumerator}-1\}$
        \cite{russell_artificial_2021}.
        The set of states is then the cartesian product
        \begin{align*}
            \facstates
        \end{align*}

            {\bf Knowledge about the system}, for example the probability of each state, is a \emph{tensor}
        \begin{align*}
            \hypercorewith \defcols \facstates \rightarrow \rr
        \end{align*}
        We represent them by diagrams with legs decorated by the variables:
        \begin{center}

            \begin{tikzpicture}[scale=0.35,thick] % , baseline = -3.5pt

    \begin{frame}{Tensor Network Decompositions}

        Necessity to overcome the curse of dimensionality.

        Definition on a hypergraph $\graph=(\nodes,\edges)$
        \begin{align*}
            \extnetat{\indexedcatvariableof{\nodes}}
            = \prod_{\edgein}\hypercoreofat{\edge}{\indexedcatvariableof{\edge}}
        \end{align*}

        Example: TT format


        Typically: Applied in a Black Box manner
        Here: We relate decompositions to the logical and probabilistic structure

    \end{frame}



    \begin{frame}{Neural function decompositions}

        \begin{block}{Neural sparsity}
            Functions of interest decompose into simple functions (neurons).
        \end{block}


        Capture by tensor networks: \emph{Basis encodings}
        of a function
        \begin{align*}
            \exformula : \bigtimes_{\catenumeratorin} [\catdimof{\catenumerator}]
            \rightarrow \bigtimes_{\selindexin} [\catdimof{\selindex}]
        \end{align*}
        is the tensor
        \begin{align*}
            \bencodingofat{\exformula}{\headvariables,\shortcatvariables}
            = \sum_{\shortcatindicesin} \onehotmapofat{\exformula(\shortcatindices)}{\headvariables}
            \otimes \onehotmapofat{\shortcatindices}{\shortcatvariables} \, .
        \end{align*}

        Basis calculus: Contraction with one-hot encoded input yields the one-hot encoded output.


    \end{frame}

    \section{Maximum entropy principles}

    \begin{frame}{Maximum entropy problem}

        Shannon Entropy: Natural measure of structure (negative information content) of a distribution.


        \begin{block}{Maximum entropy principle}
            Among those distributions reproducing certain characteristics of observed data, choose the one with the least structure (=maximum entropy).
        \end{block}


        Formally the maximum entropy problem is
        \begin{align*}
            \argmax_{\probwith} \,\,\sentropyof{\probwith}
            \quad\text{s.t.}\quad
            \contractionof{\probwith,\sencsstatwith}{\selvariable} = \meanparamwith \, .
        \end{align*}


    \end{frame}

    \begin{frame}{Generic maximum entropy distributions}

        Not all maximum entropy distributions are in the exponential family.

        Characterization by the face of the mean polytope to which the mean parameter belongs.


    \end{frame}

    \begin{frame}{Mean polytope}
        Realizable mean parameters form a convex polytope, the \emph{mean polytope} of the statistic $\sstat$:

        Exponential families can only realize the relative interior of the mean polytope.

        What are the solutions to the mean parameters on the boundary of the mean polytope?

    \end{frame}

    \begin{frame}{Summary and Outlook}
