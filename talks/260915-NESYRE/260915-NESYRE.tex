    \begin{frame}{Tensors: Natural description of factored representations}

        {\bf Factored representations}
        are collection of variables $\shortcatvariables=(\catvariableof{0},\ldots,\catvariableof{\catorder\shortminus1})$, each with a finite set of states $[\catdimof{\catenumerator}]=\{0,\ldots,\catdimof{\catenumerator}-1\}$.%\cite{russell_artificial_2021}
        The set of states is then the cartesian product
        \begin{align*}
            \facstates = \{0,\ldots,\catdimof{0}\} \times \ldots \times \{0,\ldots,\catdimof{\atomorder\shortminus1}\} \, .
        \end{align*}

            {\bf Knowledge about the system} (e.g. possibility or uncertainty of a state) is a \emph{tensor}
        \begin{align*}
            \hypercorewith \defcols \facstates \rightarrow \rr \, .
        \end{align*}
        We represent tensors by diagrams with legs decorated by the variables:
        \begin{center}
            \stantikzpic{
                \drawtensorblock{2}{-2}{3}{$\hypercore$}
                \draw (0,-3)--(0,-5) node[midway,left] {\colorlabelsize $\catvariableof{0}$};
                \draw (1.5,-3)--(1.5,-5) node[midway,left] {\colorlabelsize $\catvariableof{1}$};
                \drawtextnode{3}{-4}{\colorlabelsize $\cdots$}
                \drawvwire{4}{-3}{$\catvariableof{\catorder\shortminus1}$}
            }
        \end{center}
    \end{frame}

    \begin{frame}{Tensor Network Decompositions}

        To overcome the curse of dimensionality, tensors $\extnetat{\nodevariables}$ are decomposed into core tensors $\hypercoreofat{\edge}{\catvariableof{\edge}}$.
        Tensor networks on hypergraphs $\graph=(\nodes,\edges)$ are factorizations
        \begin{align*}
            \extnetat{\indexedcatvariableof{\nodes}}
            = \prod_{\edgein}\hypercoreofat{\edge}{\indexedcatvariableof{\edge}} \, .
        \end{align*}

            {\bf Example:} TT format
        \begin{center}
            \input{tikz/tt-format}
        \end{center}

            {\bf Beyond black-boxes:} Hypergraphs encode symbolic structure.
        \begin{itemize}
            \item Probabilistic perspective: \emph{Conditional independencies}.
            \item Neural and logical perspective: \emph{Function decompositions}.
        \end{itemize}

    \end{frame}

    \begin{frame}{Basis encodings}

        {\bf Aim:} Find tensor network decompositions based on circuit graphs.

        We define the \emph{basis encoding} of a function
        \begin{align*}
            \exformula \defcols \bigtimes_{\catenumeratorin} [\catdimof{\catenumerator}]
            \rightarrow \bigtimes_{\selindexin} [\catdimof{\selindex}]
        \end{align*}
        \vspace{-0.4cm} \\
        as the sum over one-hot encodings
        \begin{align*}
            \bencodingofat{\exformula}{\headvariables,\shortcatvariables}
            = \sum_{\shortcatindicesin} \onehotmapofat{\exformula(\shortcatindices)}{\headvariables}
            \otimes \onehotmapofat{\shortcatindices}{\shortcatvariables} \, .
        \end{align*}

            {\bf Basis calculus:} Contraction with encoded input yields the encoded output.
        \begin{align*}
            \contractionof{\bencodingofat{\exformula}{\headvariables,\shortcatvariables},\onehotmapofat{\shortcatindices}{\shortcatvariables}}{\headvariables}
            = \onehotmapofat{\exformula(\shortcatindices)}{\headvariables}
        \end{align*}

    \begin{frame}{Toy accounting example}

        \begin{block}{Toy accounting rule system}
            Assume you want to book one of two accounts $\mathrm{A1}$ and $\mathrm{A2}$ depending on a feature $\mathrm{F}$ of an invoice, based on two rules:

                \item \textcolor{\concolor}{Exactly one account must be booked.}

        Lets assume the following satisfaction rates of the toy accounting system on an invoice dataset:

            &\text{Feature $F$ implies in most cases account $A1$:} &\meanparamat{\selvariable=1} = 0.9
        \end{align*}

        \begin{center}
            \bf What is the "simplest" distribution reproducing these satsifaction rates?
        \end{center}

    \begin{frame}{Entropy: Quantifying the structure of a distribution}

        The entropy of a distribution $\probwith$ is a measure of its structure:
        \begin{align*}
            &\sentropyof{\probwith}
            &= -\sum_{\shortcatindicesin} \probat{\indexedshortcatvariables} \cdot \lnof{\probat{\indexedshortcatvariables}}
        \end{align*}
            {\bf Example:} Entropy of Bernoulli distributions \\
        \begin{center}
            \input{tikz/bernoulli-ent-plot}
        \end{center}

    \end{frame}

    \begin{frame}{Maximum entropy principle}

%        {\bf Shannon Entropy:}
%        Natural measure of structure (negative information content) of a distribution.

        \begin{block}{Maximum entropy principle}
            Among those distributions reproducing certain characteristics of observed data, choose the one with the least structure (=maximum entropy).
        \end{block}


        To simplify notation we introduce selection encodings of functions $\sstat$ as tensors with coordinates
        \begin{align*}
            \sencsstatat{\indexedshortcatvariables,\indexedselvariable}
            = \sstatcoordinateof{\selindex}(\shortcatindices) \, .
        \end{align*}

        Formally the maximum entropy problem is
        \begin{align*}
            \argmax_{\probwith} \,\,\sentropyof{\probwith}
            \quad\text{s.t.}\quad
            \contractionof{\probwith,\sencsstatwith}{\selvariable} = \meanparamwith \, .
        \end{align*}


    \end{frame}


    \begin{frame}{Exponential families are maximum entropy distributions}

        The exponential family of a statistic $\sstat$ is the set of distributions
        \begin{align*}
            \probofat{(\sstat,\canparam)}{\shortcatvariables}
            = \frac{1}{\partitionfunctionof{\canparam}} \expof{\contractionof{\canparamwith,\sencsstatwith}{\selvariable}}
        \end{align*}
        where $\canparamwith\in\parspace$.

        We find activation cores $\softactsymbolofat{\canparamat{\indexedselvariable}}{\headvariableof{\selindex}}$ to represent the members as \CompActNets{}:
        \begin{center}
            \input{tikz/exp-family-rep}
        \end{center}
            {\bf Classical result:}
        Any member of the exponential family is a solution of the maximum entropy problem to its mean parameter $\meanparamwith$.

    \end{frame}

    \begin{frame}{Outlook: Application of the maximum entropy approach}
