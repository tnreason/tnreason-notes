\section{Group fields}\label{sec:groupFields}

A group field is a vector
\begin{align*}
    \hypercoreat{\catvariable} \wcols \gengroup \rightarrow \gensemiring
\end{align*}
where $\catvariable$ enumerates the $\cardof{\gengroup}=\catdim$ elements of the group, such that the $\catindex$-th element of the group is mapped to $\hypercoreat{\indexedcatvariable}$.
Following the generalization of contractions, $\gensemiring$ is a semiring.
A complex group field is a group field for the complex sum-product semiring.


\begin{example}{Unfolding is not an isomorphism}
    Let us note that the unfolding of $[\prod_{\catenumeratorin}\catdimof{\catenumerator}]$ to $\facstates$ is not a group isomorphism.
    The group operation in $\prod_{\catenumeratorin}\catdimof{\catenumerator}$ is the integer summation
    \begin{align*}
        \catindex \circ \seccatindex =
        \catindex + \seccatindex \modspace \left(\prod_{\catenumeratorin}\catdimof{\catenumerator}\right) \, .
    \end{align*}
    In the unfolding the group operation is
    \begin{align*}
        (\catindexof{0},\ldots,\catindexof{\catorder-1}) \circ (\seccatindexof{0},\ldots,\seccatindexof{\catorder-1})
        = (\catindexof{0}+\seccatindexof{0}\modspace \catdimof{0},\ldots,\catindexof{\catorder-1}+\seccatindexof{\catorder-1}\modspace \catdimof{\catorder-1}) \, .
    \end{align*}
    Intuitively, the difference of these groups is that in the unfolded group summation carry bits are set to $0$.

\end{example}

\subsection{Subgroup encoding}

We here restrict to finite abelian groups.
A subgroup is a subset $\gensubgroup\subset\gengroup$, which is itself a group.
A coset to $\gensubgroup$ and an element $\exgroupelement\in\gengroup$ is the set
\begin{align*}
    \{\exgroupelement\circ\exsubgroupelement \wcols \exsubgroupelement\in\gensubgroup\} \, .
\end{align*}

% Dual group
The dual group $\gendualgroup$ is the collection of all characters of irreducible representations
\begin{align*}
    \characterof{k} \defcols \gengroup \rightarrow \compnums \, ,
\end{align*}
with the group operation $\characterof{k_0}\circ \characterof{k_1} (g) = \characterof{k_0}(g)\cdot\characterof{k_1}(g)$.
The annihilator of a subgroup $\gensubgroup\subset \gengroup$ is the subgroup
\begin{align*}
    \genannihilator\coloneqq
    \{\characterof{k} \wcols \characterof{k}\in\gendualgroup\ncond \uniquantwrtof{h\in \gensubgroup}{\characterofat{k}{h}=1} \} \, .
\end{align*}
By the Pontryagin duality the annihilator of the annihilator is the original subgroup \cite{serre_linear_1977, terras_fourier_1999}, that is
\begin{align*}
    \big(\genannihilator\big)^{\perp}
    = \gensubgroup \, .
\end{align*}

Since finite Abelian group are isomorphic to cartesian products of cyclic groups, we assume $\gengroup=\facgroup$.
The characters are enumerated by $\thirdcatindexof{[\catorder]}\in\facgroup$ and defined as the maps
\begin{align*}
    \characterof{\thirdcatindexof{[\catorder]}}(\shortcatindices)
    = \expof{\twopii \sum_{\catenumeratorin} \frac{\thirdcatindexof{\catenumerator}\cdot\catindexof{\catenumerator}}{\catdimof{\catenumerator}}} \, .
\end{align*}
The annihilator of a subgroup generated by $h^j_{[\catorder]}$ for $j\in[n]$ is thus defined by
\begin{align*}
    \{\characterof{\thirdcatindexof{[\catorder]}} \wcols
    \uniquantwrtof{j\in[n]}{
        \sum_{\catenumeratorin} \frac{\thirdcatindexof{\catenumerator}\cdot h^j_{\catenumerator}}{\catdimof{\catenumerator}} = 0 \modspace 1
    }\} \, .
\end{align*}

%%
We build for each $h^j_{[\catorder]}$ an edge
\begin{align*}
    \edgeof{j}
    =\{\catenumerator \wcols h^j_{\catenumerator}\neq 0\}
\end{align*}
and a tensor
\begin{align*}
    \hypercoreofat{h^j_{[\catorder]}}{\indexedcatvariableof{\edgeof{j}}}
    =\begin{cases}
         \srone & \ifspace \sum_{\catenumerator\in\edgeof{j}} \frac{\thirdcatindexof{\catenumerator}\cdot h^j_{\catenumerator}}{\catdimof{\catenumerator}} = 0 \modspace 1  \\
         \srzero & \elsetext
    \end{cases} \, .
\end{align*}
Then we have for the subset encoding of $\genannihilator\subset\gendualgroup$ (where $\gendualgroup$ is enumerated by $\shortcatindicesin$) that
\begin{align*}
    \onehotmapofat{\genannihilator}{\shortcatvariables}
    = \contractionof{
        \{\hypercoreofat{h^j_{[\catorder]}}{\indexedcatvariableof{\edgeof{j}}} \wcols j\in[n]\}
    }{\shortcatvariables} \, .
\end{align*}

%% Representation by 
To find a tensor network decomposition of $\hypercoreofat{h^j_{[\catorder]}}{\catvariableof{\edgeof{j}}}$ we define the smallest common multiple of the dimensions $\{\catdimof{\catenumerator}\wcols\catenumeratorin\}$ by $\catdim$ and notice that the condition for non-vanishing coordinates is equal to
\begin{align*}
    \sum_{\catenumerator\in\edgeof{j}} \frac{\catdim}{\catdimof{\catenumerator}} \cdot \thirdcatindexof{\catenumerator}\cdot h^j_{\catenumerator}
    = 0 \modspace \catdim
\end{align*}
Since $\catdim$ is the smallest common multiple, $\frac{\catdim}{\catdimof{\catenumerator}}$ is an integer and we define for $\catenumeratorin$ functions
\begin{align*}
    \secexfunctionof{\catenumerator} \wcols [\catdim] \times [\catdim] \rightarrow [\catdim]
    \quad , \quad
    \secexfunctionof{\catenumerator}(\headindexof{\catenumerator},\catindexof{\catenumerator})
    = \headindexof{\catenumerator} + \frac{\catdim}{\catdimof{\catenumerator}} \cdot \catindexof{\catenumerator} \modspace \catdim \, .
\end{align*}

\begin{example}\label{exa:cosetsCyclicTwoProduct} % WH case

    The condition becomes
    \begin{align*}
        \sum_{\catenumerator\in\edgeof{j}} \thirdcatindexof{\catenumerator} = 0 \modspace 2
    \end{align*}
    Notice, that this is captured by the generalized exactly or connective and thus
    \begin{align*}
        \hypercoreofat{h^j_{[\catorder]}}{\catvariableof{\edgeof{j}}}
        = \bigoplus [\catvariableof{\edgeof{j}}] \, .
    \end{align*}
    Using the associativity of the exactly or connective, one can decompose the tensor in an arbitrary tree format with hidden ranks $2$.
